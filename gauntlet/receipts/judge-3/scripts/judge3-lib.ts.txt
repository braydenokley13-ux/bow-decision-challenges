export async function openRun(page: Page, classCode: string, seat: string) {
  const card = await F.seatOnRoster(page, classCode, seat);
  await F.signIn(page, { ...card, classCode });
  await page.getByRole("link", { name: /^(Start|Carry on)$/ }).click();
  await F.startIfConfirmAsked(page);
  await F.chooseSeasonIfOffered(page);
  await F.stepPastTheDeal(page);
  return card;
}

/** Wrong answers until the product hands the answer over, then take it. */
async function takeTheAnswer(page: Page, scope: string, label: RegExp | string) {
  for (let i = 1; i <= 8; i += 1) {
    const give = page.getByRole("button", { name: "Show the answer and keep going" });
    if (await give.count()) { await give.first().click(); return i - 1; }
    await page.getByLabel(label).fill(String(1000 + i * 137));
    await page.locator(scope).getByRole("button", { name: "Check" }).click();
    await page.waitForTimeout(220);
  }
  throw new Error(`no answer offered at ${scope}`);
}

/** A student who does not think. */
export async function playFast(page: Page, log: (s: string) => void) {
  await page.getByRole("button", { name: "Check the order" }).click();
  await page.waitForTimeout(250);
  for (let i = 0; i < 8; i += 1) {
    if (await page.getByRole("button", { name: "Show the answer and keep going" }).count()) break;
    await page.getByRole("button", { name: "Check it again" }).click();
    await page.waitForTimeout(250);
  }
  await page.getByRole("button", { name: "Show the answer and keep going" }).click();
  await page.getByRole("button", { name: "Keep going" }).click();
  await page.locator(".place-card").nth(0).getByRole("button", { name: "Choose this setup" }).click();
  await takeTheAnswer(page, ".chosen-total", /What the .* costs Avery/);
  await page.getByRole("button", { name: "Build the plan" }).click();
  await takeTheAnswer(page, ".calculation", F.PLAN_STEP.countOn);
  await page.getByRole("button", { name: F.PLAN_STEP.toBonuses }).click();
  for (const i of [0, 1]) await page.locator(".bet").nth(i).getByRole("button", { name: F.PLAN_STEP.countBonus }).click();
  await page.getByRole("button", { name: F.PLAN_STEP.toCommitted }).click();
  await takeTheAnswer(page, ".calculation", F.PLAN_STEP.committed);
  await page.getByRole("button", { name: F.PLAN_STEP.toPlan }).click();

  const FORWARD = /^(Save |Lock |Next|Keep going|Play |Explain|Turn in|Continue|Reserve|Wait and|Take the|Keep the|Yes —|No —|Build |Check this plan|Start|Go |Send|Finish|Move on|Plan without|Count it)/i;
  const SKIP = /^(Back|See where|Where Avery lives|Food, phone|1Money|2The two|3Money|Decrease|Increase|Move .*(earlier|later)|Not you|Sign out|Read|Hide|Selected:|Show me one step)/i;
  const seen = new Map<string, number>();
  for (let step = 0; step < 140; step += 1) {
    await page.waitForTimeout(100);
    if (await page.getByRole("heading", { name: "Your plan is with your teacher." }).count()) { log(`fast: submitted at ${step}`); return; }
    const heading = (await page.locator("main h1, main h2").first().innerText().catch(() => "")).replace(/\s+/g, " ").slice(0, 45);
    if (await page.locator(".interview__stats button").count()) {
      const tiles = page.locator(".interview__stats button");
      const figs: string[] = [];
      for (const i of [0, 1]) { await tiles.nth(i).click(); figs.push((await tiles.nth(i).locator(".money").innerText()).trim()); }
      const junk = `asdf asdf ${figs[0]} asdf. qwer qwer ${figs[1]} qwer.`;
      await page.getByLabel("Two to four sentences").fill(junk);
      log(`fast: wrote ${JSON.stringify(junk)}`);
      await page.getByRole("button", { name: "Turn in my plan" }).click(); continue;
    }
    const give = page.getByRole("button", { name: "Show the answer and keep going" });
    if (await give.count()) { log(`fast: [${heading}] took the answer`); await give.first().click(); continue; }
    const nothing = page.getByRole("button", { name: /— nothing this season$/ });
    if (await nothing.count()) { await nothing.first().click(); continue; }
    const btns = page.locator("main button:not([disabled])");
    const read = async () => btns.evaluateAll((els) => els.map((e) => ((e as HTMLElement).getAttribute("aria-label") || (e as HTMLElement).textContent || "").replace(/\s+/g, " ").trim()));
    let labels = await read();
    const inputs = page.locator("main input:not([type=checkbox]):not([type=radio]):not([type=hidden])");
    let typed = "";
    for (let i = 0; i < (await inputs.count()); i += 1) {
      const el = inputs.nth(i);
      if (!(await el.isVisible().catch(() => false))) continue;
      const v = await el.inputValue().catch(() => "");
      if (v === "" || v === "0") { await el.fill(String(300 + step * 17)); typed += `${300 + step * 17} `; }
    }
    if (typed) { labels = await read(); const ci = labels.findIndex((l) => /^Check$/i.test(l)); if (ci >= 0) { await btns.nth(ci).click(); continue; } }
    const cands = labels.map((l, i) => ({ l, i })).filter((c) => c.l && !SKIP.test(c.l));
    const pick = cands.filter((c) => FORWARD.test(c.l)).find((c) => (seen.get(`${heading}|${c.l}`) ?? 0) < 3)
      ?? cands.find((c) => (seen.get(`${heading}|${c.l}`) ?? 0) < 3) ?? cands.at(-1);
    if (!pick) { log(`fast: stuck at ${heading}`); return; }
    seen.set(`${heading}|${pick.l}`, (seen.get(`${heading}|${pick.l}`) ?? 0) + 1);
    log(`fast ${step} [${heading}] -> ${pick.l.slice(0, 45)}`);
    await btns.nth(pick.i).click({ timeout: 8000 }).catch(() => log(`fast: click failed at ${heading}`));
  }
}

/**
 * Careful, unaided arithmetic and a real explanation — attached to a bad plan: the most
 * expensive room, both conditional bonuses counted as income, nothing kept back.
 */
export async function playReasonedButBad(page: Page, log: (s: string) => void) {
  const plan: PlanContext = { setupId: "gym-sublet", countCompletion: true, countOutcome: true };
  await F.completeSetupStage(page, 0);                       // right first time
  await F.completeWorkingCalcs(page, { attendance: true, showcase: true });
  // Everything into the course and rides; nothing kept back.
  await F.setAmount(page, "Sports-media course", "1200");
  await F.setAmount(page, "Backup money", "0");
  await restCard(page, "flexibleCash").click();
  const nothing = page.getByRole("button", { name: /— nothing this season$/ });
  for (let g = 0; g < 3 && (await nothing.count()) > 0; g += 1) await nothing.first().click();
  await page.getByRole("button", { name: "Save this version" }).click();
  // The fallback check: both bonuses removed. Free it from rides, keep the course whole.
  await savePlan(page, "fallback", plan);
  await F.playSeasonWeeks(page, { claim: 0, reason: 1, deposit: false });
  await F.passWeek5Calculation(page, String(week5TotalFor(plan)));
  await savePlan(page, "week5-first-response", plan);
  await F.decideOpportunity(page, { clinics: true, countBonus: false });
  const landed = { ...plan, clinics: true, countCompletionFinal: false };
  await savePlan(page, "final", landed);
  // The last check: take the missing amount out of the row the student is willing to lose.
  for (let g = 0; g < 6; g += 1) {
    if (await page.getByRole("button", { name: "Save the last check" }).count()) break;
    const take = page.getByRole("button", { name: /^Take \$[\d,]+ from / });
    if (!(await take.count())) break;
    await take.first().click();
    await page.waitForTimeout(150);
  }
  if (await page.getByRole("button", { name: "Save the last check" }).count()) {
    await page.getByRole("button", { name: "Save the last check" }).click();
  }
  await F.readWeek8Resolution(page);
  await F.submitDefense(
    page,
    "I put Avery in the sublet because the trip is only an hour a week and that time is worth more than the money. I counted the bonuses because Avery has made every session so far, and I kept the whole course payment because that is the thing the season is for. What I gave up was backup money.",
  );
  await F.waitForDelivery(page);
  log("reasoned-bad: submitted");
}

/** The competent run: cheapest room, no conditional money counted, a reserve kept. */
export async function playCompetent(page: Page, log: (s: string) => void) {
  const plan: PlanContext = { setupId: "cousin-room" };
  await F.completeSetupStage(page, 2);
  await F.completeWorkingCalcs(page);
  await savePlan(page, "working", plan);
  await F.playSeasonWeeks(page, { claim: 0, reason: 0, deposit: false });
  await F.passWeek5Calculation(page, String(week5TotalFor(plan)));
  await savePlan(page, "week5-first-response", plan);
  await F.decideOpportunity(page, { clinics: false, countBonus: false });
  await savePlan(page, "final", { ...plan, clinics: false, countCompletionFinal: false });
  await F.readWeek8Resolution(page);
  await F.submitDefense(
    page,
    "I kept Avery at the cousin's room because the rent is the one cost I could get almost to nothing, and I left both bonuses out of the total because neither of them is promised. When Week 5 took the showcase away I cut the rides money rather than the course, because the course is the thing Avery is saving for and the rides can go back to walking.",
  );
  await F.waitForDelivery(page);
  log("competent: submitted");
}

/** Starts, builds the opening plan, and walks away without turning anything in. */
export async function playAbandoned(page: Page, log: (s: string) => void) {
  await F.completeSetupStage(page, 2);
  await F.completeWorkingCalcs(page);
  await savePlan(page, "working", { setupId: "cousin-room" });
  await page.waitForTimeout(1200);
  log("abandoned: stopped after the opening plan");
}
