import { test, expect } from "@playwright/test";
import * as F from "./flow";

test("setup total gate: what happens to a student who keeps getting it wrong", async ({ page, request }) => {
  test.setTimeout(200_000);
  const cls = await F.createClass(request, "Judge 3 gate");
  await F.gotoFreshChallenge(page);
  await F.enterChallenge(page, { classCode: cls.code, seatCode: "1" });
  await F.rankPlacesCorrectly(page);
  await page.locator(".place-card").nth(2).getByRole("button", { name: "Choose this setup" }).click();
  const field = page.getByLabel(/What the .* costs Avery/);
  for (let attempt = 1; attempt <= 8; attempt += 1) {
    await field.fill(String(100 * attempt + 7));
    await page.locator(".chosen-total").getByRole("button", { name: "Check" }).click();
    await page.waitForTimeout(250);
    const txt = (await page.locator(".chosen-total").innerText()).replace(/\s+/g, " ");
    const btns = await page.locator("main button:not([disabled])").evaluateAll((els) =>
      els.map((e) => ((e as HTMLElement).getAttribute("aria-label") || (e as HTMLElement).textContent || "").replace(/\s+/g, " ").trim()).filter(Boolean));
    console.log(`ATTEMPT ${attempt}: ${txt.slice(0, 400)}`);
    console.log(`  BUTTONS: ${JSON.stringify(btns)}`);
  }
});
